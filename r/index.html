    R guide¶

    library(ewe)
m <- ewe_library_load("tampa-bay")         # or ewe_open("my_model.eiixml" / "my_model.ewemdb")
path <- ewe_ecopath(m)
subset(path$groups, ee > 1)

sim <- ewe_ecosim(m, years = 30, effort = c("Bait Trawl" = 0.5))
head(ewe_wide(sim$groups, "biomass"))      # time x group
ewe_close(m)

    The package starts the EwE engine (ewe-engine serve) as a child process and speaks the same JSON protocol as the Python SDK, so both return the same numbers.

    Install the package and an engine bundle as described in Install. Requires jsonlite and processx; charts need ggplot2.

    Network analysis and ecological indicators¶

    m <- ewe_library_load("anchovy-bay")
na <- ewe_network_analysis(m)              # EwE's Network Analysis plug-in, on the Ecopath model
na$summary                                 # throughput, Finn's cycling index, ascendency, TE, ...
ewe_mti_matrix(na)                         # mixed trophic impacts (rows impact columns)
head(ewe_ecosim_network_analysis(m, years = 20)$indices)   # network indices per time step

ind <- ewe_indicators(m, ecosim = TRUE)    # Ecological Indicators plug-in
ind$ecopath                                # indicator, category, value
head(ind$ecosim)                           # time step x indicator

    Plotting¶

    With ggplot2 installed (install.packages("ggplot2"); it is in Suggests), ewe_plot_*() functions return ggplot objects, so they print as charts and take more layers with +:

    sim <- ewe_ecosim(m)
ewe_plot_ecosim(sim, relative = TRUE)                # small multiples past eight groups
ewe_plot_ecosim(sim, "catch", groups = c("Cod", "Anchovy"))
ewe_plot_fleets(sim)                                 # landings by fleet, stacked
ewe_plot_compare(list(baseline = sim, "seiners x2" = ewe_ecosim(m, effort = c(Seiners = 2))))
ewe_plot_fit(ewe_time_series(m, 2), list(before = sim, after = after))
ewe_plot_fit_iterations(fit)
ewe_plot_monte_carlo(ewe_monte_carlo(m, trials = 50))
ewe_plot_mse(ewe_mse(m), "catch")
ewe_plot_msy(ewe_msy(m, "Anchovy")); ewe_plot_fmsy(ewe_fmsy(m))
ewe_plot_ecospace_map(sp, "Cod"); ewe_plot_ecospace_map(sp); ewe_plot_ecospace(sp)
ewe_plot_foodweb(m); ewe_plot_trophic_levels(ewe_ecopath(m))
ewe_plot_mti(ewe_network_analysis(m))
ewe_plot_theme("dark")                               # dark palette for every chart after this

    Building and editing models¶

    doc <- ewe_new_model("Toy bay", area = 100)
doc <- ewe_add_group(doc, "Fish", biomass_area = 2, pb = 0.8, qb = 4)
doc <- ewe_add_group(doc, "Zooplankton", pb = 30, qb = 100, ee = 0.9)
doc <- ewe_add_group(doc, "Phytoplankton", "producer", biomass_area = 15, pb = 120)
doc <- ewe_add_group(doc, "Detritus", "detritus", biomass_area = 50)
doc <- ewe_set_diet(doc, "Fish", c(Zooplankton = 1))
doc <- ewe_set_diet(doc, "Zooplankton", c(Phytoplankton = 0.8, Detritus = 0.2))
doc <- ewe_add_fleet(doc, "Trawl")
doc <- ewe_set_catch(doc, "Trawl", "Fish", landings = 0.3)
doc <- ewe_add_ecosim_scenario(doc, "Baseline", years = 20)
ewe_validate(doc)
ewe_save_model(doc, "toy.eiixml")

m <- ewe_open("Tampa_Bay.eiixml")
doc <- ewe_document(m)
doc <- ewe_set_group(doc, "Pin Fish", pb = 2.5)
doc <- ewe_remove_group(doc, "Catfish")
ewe_apply(m, doc)        # run the edit; the file is untouched
ewe_ecopath(m)
ewe_save(m, "Tampa_Bay_edited.eiixml")

    NA means "not entered". ewe_doc_table(doc, "groups") and ewe_diet_matrix(doc) give data.frames; ewe_update_groups, ewe_set_diet_matrix and ewe_set_catch_table take them back. Invalid documents signal an ewe_validation_error carrying an errors data.frame.

    Multi-stanza groups, Ecosim inputs and time series to fit to:

    doc <- ewe_add_stanza(doc, "Herring", data.frame(group = c("Herring juvenile", "Herring adult"),
                                                 age_start = c(0, 24), mortality = c(1.5, 0.7)))
doc <- ewe_calculate_stanzas(doc)$doc   # B and Q/B of the non-leading stages
doc <- ewe_set_effort(doc, "Baseline", "Trawl", c(1, 1, 2))   # per year, relative to Ecopath effort
doc <- ewe_set_vulnerabilities(doc, "Baseline", data.frame(predator = "Fish", prey = "Zooplankton", vulnerability = 3))
doc <- ewe_add_time_series(doc, "Survey", "Fish index", "biomass_relative", c(2, NA, 1.8), group = "Fish")

    The Python guide explains the fields; the rules are the same. ewe_fit() starts from the scenario's own vulnerabilities (reset_vulnerabilities = TRUE starts from 2). ewe_from_ewemdb() and ewe_to_ewemdb() refuse to replace an existing file unless overwrite = TRUE.

    Engine bundle (installation)¶

    ewe_install_engine("ewe-bundle-osx-arm64.tar.gz") installs a self-contained bundle (engine, Access converter and model library, .NET included) into the package's user data folder; after that nothing else is needed. Bundles are built by python/tools/build_wheel.py <platform>.

    Python and R names¶

    Every Python method has an R function taking the model first (m.ecosim(...) is ewe_ecosim(m, ...)). The ones that differ:

    Python R

    sim.biomass, sim.wide("catch") ewe_wide(sim$groups, "biomass")
    mse.wide("biomass", "q05") ewe_mse_wide(mse, "biomass", "q05")
    msy.yield_curve() ewe_yield_curve(msy)
    na.stats, na.mti_matrix() na$summary, ewe_mti_matrix(na)
    ind.values ind$ecopath
    sim.plot.biomass(...), ts.plot.fit(sim), m.plot.foodweb() ewe_plot_ecosim(sim, ...), ewe_plot_fit(ts, sim), ewe_plot_foodweb(m)
    ewe.convert.mdb_to_eiixml, eiixml_to_mdb ewe_from_ewemdb, ewe_to_ewemdb
    m.document(), doc.set_group(...) ewe_document(m), doc <- ewe_set_group(doc, ...)
